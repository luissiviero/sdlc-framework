  ```python
  import json, os, sys
  from pathlib import Path
  sys.path.insert(0, str(Path(sys.argv[1]) / "plugin"))
  from gate import checks
  root = Path(sys.argv[1])
  cases = {
      "both green": 'python -c "print(1)" && python -c "print(2)"',
      "first red": 'python -c "import sys; print(1); sys.exit(3)" && python -c "print(2)"',
      "second red": 'python -c "print(1)" && python -c "import sys; print(2); sys.exit(4)"',
  }
  print("os.name:", os.name, "| sys.platform:", sys.platform)
  for name, cmd in cases.items():
      print(f"--- run_command [{name}]")
      print(json.dumps(checks.run_command(cmd, root, 60)))
  ctx = checks.GateContext(
      root=root, change_dir=root, phase="c", status=None, diff=None, human_gate=False,
      config={"commands": {"build": 'python -c "print(0)"', "test": 'python -c "print(0)"',
                           "lint": cases["both green"]}, "gate": {"command_timeout": 60}},
  )
  for label, lint in (("both green", cases["both green"]), ("second red", cases["second red"])):
      ctx.config["commands"]["lint"] = lint
      print(f"--- run_targets -> judge_targets, lint chained, {label}")
      print(json.dumps(checks.judge_targets(checks.run_targets(ctx), "runner").as_dict(), default=str))
  ```
